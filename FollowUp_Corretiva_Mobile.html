create table if not exists public.followup_om_itens ( id uuid primary key default gen_random_uuid(), om_id text not null, ordem text, codigo_material text, pn text, descricao text not null, quantidade numeric(14,2) not null default 1, unidade text, status text not null default 'Pendente', observacao text, created_at timestamptz not null default now(), updated_at timestamptz not null default now() ); -- Para quem já criou a tabela pela versão anterior: alter table public.followup_om_itens add column if not exists status text not null default 'Pendente'; update public.followup_om_itens set status = 'Pendente' where status is null or status not in ('Pendente','Recebido'); do $$ begin if not exists ( select 1 from pg_constraint where conname = 'followup_om_itens_status_check' and conrelid = 'public.followup_om_itens'::regclass ) then alter table public.followup_om_itens add constraint followup_om_itens_status_check check (status in ('Pendente','Recebido')); end if; end $$; create index if not exists idx_followup_om_itens_om_id on public.followup_om_itens (om_id); create index if not exists idx_followup_om_itens_ordem on public.followup_om_itens (ordem); create index if not exists idx_followup_om_itens_pn on public.followup_om_itens (pn); create index if not exists idx_followup_om_itens_status on public.followup_om_itens (status); create or replace function public.followup_om_itens_set_updated_at() returns trigger language plpgsql as $$ begin new.updated_at = now(); return new; end; $$; drop trigger if exists trg_followup_om_itens_updated_at on public.followup_om_itens; create trigger trg_followup_om_itens_updated_at before update on public.followup_om_itens for each row execute function public.followup_om_itens_set_updated_at(); alter table public.followup_om_itens enable row level security; do $$ begin if not exists ( select 1 from pg_policies where schemaname='public' and tablename='followup_om_itens' and policyname='followup_om_itens_public_all' ) then create policy followup_om_itens_public_all on public.followup_om_itens for all to anon, authenticated using (true) with check (true); end if; end $$; grant select, insert, update, delete on public.followup_om_itens to anon, authenticated;
